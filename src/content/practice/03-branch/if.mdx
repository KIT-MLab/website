<Exercise
  id="practice-03-if-5"
  kind="build"
  level={3}
  name="勉強時間の表示"
  stdin={`135`}
  tests={[
    { kind: 'stdout', stdin: `135\n` },

    { kind: 'stdout', stdin: `60\n` },

    { kind: 'stdout', stdin: `0\n` },
    { kind: 'stdout', stdin: `600\n` },
  ]}
  hints={['60分以上のときは、何時間かと、残りの何分かに分けます。第2章3節の `//` と `%` を使います。', '表示に変数の値を入れる書き方は、第1章の f文字列です。']}
  syntax={['input', 'compare', 'if', 'operators', 'print']}

</Output>
