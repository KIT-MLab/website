hours  = np.array([2.0, 3.0, 5.0, 6.0, 8.0, 9.0, 6.0, 6.5, 8.5, 5.5, 7.0, 8.0, 2.5, 0.5, 3.0, 3.0, 8.0, 8.5, 0.5, 4.5, 7.5, 1.5, 7.5, 1.5, 4.5, 7.5, 3.0, 3.5, 3.0, 6.5])

passed = (scores >= 60) * 1

rng = np.random.default_rng(0)
idx = rng.permutation(30)
train_idx = idx[:24]
test_idx = idx[24:]

h_mean = hours[train_idx].mean()
h_std = hours[train_idx].std()
s_mean = sleep[train_idx].mean()
s_std = sleep[train_idx].std()

X_train = np.array([(hours[train_idx] - h_mean) / h_std, (sleep[train_idx] - s_mean) / s_std]).T
X_test = np.array([(hours[test_idx] - h_mean) / h_std, (sleep[test_idx] - s_mean) / s_std]).T
y_train = passed[train_idx]
y_test = passed[test_idx]

def sigmoid(z):
    return 1 / (1 + np.exp(-z))

def loss(X, y, w, b):
    p = sigmoid(X @ w + b)
    return np.mean(-(y * np.log(p) + (1 - y) * np.log(1 - p)))

def grad_w(X, y, w, b):
    p = sigmoid(X @ w + b)
    return X.T @ (p - y) / len(y)

def grad_b(X, y, w, b):
    p = sigmoid(X @ w + b)
    return np.mean(p - y)

w = np.array([0.0, 0.0])
b = 0.0

for i in range(300):
    gw = grad_w(X_train, y_train, w, b)
    gb = grad_b(X_train, y_train, w, b)
    w = w - 0.5 * gw
    b = b - 0.5 * gb

loss_after = loss(X_train, y_train, w, b)
pred_test = (sigmoid(X_test @ w + b) >= 0.5) * 1

accuracy = (pred_test == y_test).mean()
ones = y_test.sum()
zeros = len(y_test) - ones
if ones >= zeros:
    baseline = ones / len(y_test)
else:
    baseline = zeros / len(y_test)

hit = np.sum((pred_test == 1) & (y_test == 1))
miss = np.sum((pred_test == 0) & (y_test == 1))
false_alarm = np.sum((pred_test == 1) & (y_test == 0))
correct_reject = np.sum((pred_test == 0) & (y_test == 0))

print(round(loss_before, 3))
print(round(loss_after, 3))
print(np.round(w, 2))

{/* 説明 */}

8〜11行目の分け方は第15章1節と同じです。13〜19行目で、訓練データとテストデータの両方を標準化します（第13章5節）。

このとき、平均と標準偏差は訓練データのものだけを使います。実際に使うときは、新しい人のデータは、その人が来るまで手元にありません。そのため、新しい人を入れた平均は、あらかじめ求められないのです。

テストデータは、その新しい人の代わりです。

`loss`・`grad_w`・`grad_b`は、訓練データとテストデータで別のデータを渡せるように、`X`と`y`も引数にしました。

49行目は、テストデータの6人を、確率0.5以上なら1、そうでなければ0と予測します。

出力の8行は、順に、学習前の損失、学習後の損失、`w`、`b`、テストデータの予測、正解率、ベースライン、4つの場合の数です。

損失、正解率とベースライン、4つの場合の数の式は、これまでの節と同じです。テストデータの正解率`0.83`は、ベースラインの`0.5`を上回りました。テストデータは6人だけなので、分け方を変えると正解率は大きく動き、ベースラインと同じか、下回ることもあります。

この節の最初のコードの3〜6行目のデータを、そのまま使ってください。データは、30人の勉強時間・睡眠時間・点数と、60点以上を1にした合否です。

入力の種で30人を並べ替え、先頭の24人を訓練データ、残りの6人をテストデータに分けます。勉強時間と睡眠時間は、訓練データの平均と標準偏差で、訓練データとテストデータの両方を標準化します。

標準化した2つの入力で、ロジスティック回帰を学習してください。`w`は`[0, 0]`、`b`は`0`から始め、学習率は0.5、くり返しは300回です。

テストデータの6人について、確率が0.5以上なら合格（1）、そうでなければ不合格（0）と予測します。この予測から、正解率とベースライン、4つの場合の人数を求めてください。正解率とベースラインは`round()`で小数第2位まで丸めます。

- 1行目: 乱数の種（整数）
</Input>

<Output>
1行目に正解率、2行目にベースラインを表示します。3行目に、当たり・見逃し・空振り・正しく不合格の人数を、この順に空白で区切って表示します。
